<script>
    let ratingChartInstance = null;
    let heatmapChartInstance = null;
    const defaultAvatar = `data:image/svg+xml;utf8,<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 24 24" fill="%23cbd5e0"><path d="M12 12c2.21 0 4-1.79 4-4s-1.79-4-4-4-4 1.79-4 4 1.79 4 4 4zm0 2c-2.67 0-8 1.34-8 4v2h16v-2c0-2.66-5-4-8-4z"/></svg>`;
    async function fetchProfileData() {
        const handle = document.getElementById('handleInput').value.trim();
        const errorDiv = document.getElementById('errorMessage');
        const dashboard = document.getElementById('dashboard');
        if (!handle) return;
        errorDiv.style.display = 'none';
        dashboard.style.display = 'none';
        try {
            const [infoRes, statusRes, ratingRes] = await Promise.all([
                fetch(`https://codeforces.com/api/user.info?handles=${handle}`),
                fetch(`https://codeforces.com/api/user.status?handle=${handle}`),
                fetch(`https://codeforces.com/api/user.rating?handle=${handle}`)
            ]);
            const infoData = await infoRes.json();
            const statusData = await statusRes.json();
            const ratingData = await ratingRes.json();
            if (infoData.status !== "OK" || statusData.status !== "OK" || ratingData.status !== "OK") {
                throw new Error("API error");
            }
            const user = infoData.result[0];
            updateProfileCard(user);
            processSubmissions(statusData.result);
            processRatingHistory(ratingData.result);
            dashboard.style.display = 'block';
        } catch (error) {
            console.error(error);
            errorDiv.style.display = 'block';
        }
    }
    function updateProfileCard(user) {
        const avatarImg = document.getElementById('userAvatar');
        let rawPhoto = user.titlePhoto || user.avatar || '';
        if (rawPhoto.startsWith('//')) {
            rawPhoto = 'https:' + rawPhoto;
        }
        avatarImg.src = rawPhoto || defaultAvatar;
        avatarImg.onerror = function() {
            this.src = defaultAvatar;
            this.onerror = null; 
        };
        document.getElementById('userHandle').innerText = user.handle;
        document.getElementById('userRank').innerText = `${user.rank || 'Unrated'} (Max: ${user.maxRank || 'Unrated'})`;
        document.getElementById('currentRating').innerText = user.rating || 'Unrated';
        document.getElementById('highestRating').innerText = user.maxRating || 'Unrated';
    }
    function processSubmissions(submissions) {
        const solvedProblems = new Set();
        const tagCounts = {};
        const ratingCounts = {};
        submissions.forEach(sub => {
            if (sub.verdict === "OK") {
                const problemId = `${sub.problem.contestId}-${sub.problem.index}`;
                if (!solvedProblems.has(problemId)) {
                    solvedProblems.add(problemId);
                    if (sub.problem.tags) {
                        sub.problem.tags.forEach(tag => {
                            tagCounts[tag] = (tagCounts[tag] || 0) + 1;
                        });
                    }
                    if (sub.problem.rating) {
                        const r = sub.problem.rating;
                        ratingCounts[r] = (ratingCounts[r] || 0) + 1;
                    }
                }
            }
        });
        document.getElementById('totalSolved').innerText = solvedProblems.size;
        renderHeatmap(ratingCounts);
        renderFullTagsList(tagCounts);
    }
    function renderFullTagsList(tagCounts) {
        const list = document.getElementById('tagList');
        list.innerHTML = '';
        const sortedTags = Object.entries(tagCounts).sort((a, b) => b[1] - a[1]);
        if (sortedTags.length === 0) {
            list.innerHTML = '<li class="tag-item"><span>No problem tag data found.</span></li>';
            return;
        }
        sortedTags.forEach(([tag, count]) => {
            const li = document.createElement('li');
            li.className = 'tag-item';
            li.innerHTML = `<span>${tag}</span><strong>${count} problems</strong>`;
            list.appendChild(li);
        });
    }
    function renderHeatmap(ratingCounts) {
        const ctx = document.getElementById('heatmapChart').getContext('2d');
        if (heatmapChartInstance) heatmapChartInstance.destroy();
        const sortedRatings = Object.keys(ratingCounts).sort((a, b) => a - b);
        const dataValues = sortedRatings.map(r => ratingCounts[r]);
        heatmapChartInstance = new Chart(ctx, {
            type: 'bar',
            data: {
                labels: sortedRatings,
                datasets: [{
                    label: 'Problems Solved',
                    data: dataValues,
                    backgroundColor: '#4299e1',
                    borderRadius: 4
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                plugins: { legend: { display: false } },
                scales: {
                    x: { title: { display: true, text: 'Difficulty Rating' } },
                    y: { title: { display: true, text: 'Count' }, beginAtZero: true }
                }
            }
        });
    }
    function processRatingHistory(ratingData) {
        const ctx = document.getElementById('ratingChart').getContext('2d');
        if (ratingChartInstance) ratingChartInstance.destroy();
        const labels = ratingData.map((_, index) => `Round ${index + 1}`);
        const dataValues = ratingData.map(contest => contest.newRating);
        ratingChartInstance = new Chart(ctx, {
            type: 'line',
            data: {
                labels: labels,
                datasets: [{
                    label: 'Rating',
                    data: dataValues,
                    borderColor: '#3182ce',
                    backgroundColor: 'rgba(49, 130, 206, 0.1)',
                    borderWidth: 2,
                    pointRadius: 2,
                    fill: true,
                    tension: 0.1
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                plugins: {
                    legend: { display: false },
                    tooltip: {
                        callbacks: {
                            title: (context) => ratingData[context[0].dataIndex].contestName
                        }
                    }
                },
                scales: {
                    y: { title: { display: true, text: 'Rating' } }
                }
            }
        });
    }
